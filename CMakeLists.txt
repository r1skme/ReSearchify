cmake_minimum_required(VERSION 3.20)
project(ReSearchify VERSION 2.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RESEARCHIFY_STATIC_RUNTIME "Use the static MSVC runtime for a portable executable" ON)
option(RESEARCHIFY_LTO "Enable release link-time optimization when supported" OFF)
option(RESEARCHIFY_BUILD_BENCHMARKS "Build the standalone benchmark executable" ON)
if(MSVC AND RESEARCHIFY_STATIC_RUNTIME)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(RESEARCHIFY_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
    if(NOT ipo_supported)
        message(FATAL_ERROR "LTO requested but not supported: ${ipo_error}")
    endif()
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
option(RESEARCHIFY_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)

find_package(Threads REQUIRED)
add_library(researchify_core src/engine.cpp src/query.cpp src/storage.cpp src/disk.cpp src/json.cpp src/api.cpp)
target_include_directories(researchify_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(researchify_core PUBLIC Threads::Threads)

function(researchify_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(RESEARCHIFY_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()
researchify_warnings(researchify_core)

add_executable(researchify src/main.cpp)
set_target_properties(researchify PROPERTIES OUTPUT_NAME rs)
target_link_libraries(researchify PRIVATE researchify_core)
researchify_warnings(researchify)

include(CTest)
if(BUILD_TESTING)
    foreach(suite IN ITEMS json disk v2 process)
        add_executable(cs_${suite}_tests tests/${suite}_tests.cpp)
        target_link_libraries(cs_${suite}_tests PRIVATE researchify_core)
        researchify_warnings(cs_${suite}_tests)
        add_test(NAME ${suite} COMMAND cs_${suite}_tests)
        set_tests_properties(${suite} PROPERTIES TIMEOUT 240)
    endforeach()
    add_executable(cs_tests tests/engine_tests.cpp)
    target_link_libraries(cs_tests PRIVATE researchify_core)
    researchify_warnings(cs_tests)
    add_test(NAME engine COMMAND cs_tests)
    set_tests_properties(engine PROPERTIES TIMEOUT 180)

    add_executable(cs_query_tests tests/query_tests.cpp)
    target_link_libraries(cs_query_tests PRIVATE researchify_core)
    researchify_warnings(cs_query_tests)
    add_test(NAME query COMMAND cs_query_tests)
    set_tests_properties(query PROPERTIES TIMEOUT 60)

    add_executable(cs_storage_tests tests/storage_tests.cpp)
    target_link_libraries(cs_storage_tests PRIVATE researchify_core)
    researchify_warnings(cs_storage_tests)
    add_test(NAME storage COMMAND cs_storage_tests)
    set_tests_properties(storage PROPERTIES TIMEOUT 120)
    add_test(NAME cli COMMAND ${CMAKE_COMMAND}
        -DEXE=$<TARGET_FILE:researchify>
        -DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-test-data
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.cmake)
    set_tests_properties(cli PROPERTIES TIMEOUT 60)
    add_test(NAME api_cli COMMAND ${CMAKE_COMMAND}
        -DEXE=$<TARGET_FILE:researchify>
        -DWORK=${CMAKE_CURRENT_BINARY_DIR}/api-cli-test-data
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/api_cli_tests.cmake)
    set_tests_properties(api_cli PROPERTIES TIMEOUT 90)
endif()

if(RESEARCHIFY_BUILD_BENCHMARKS)
    add_executable(cs_bench bench/benchmark.cpp)
    target_link_libraries(cs_bench PRIVATE researchify_core)
    researchify_warnings(cs_bench)
endif()

install(TARGETS researchify RUNTIME DESTINATION bin)

install(FILES README.md README.ru.md DESTINATION share/researchify)
install(DIRECTORY skills/ DESTINATION share/researchify/skills)
